\section{Kitaev Model}
\begin{frame}{Kitaev Model: an Introduction}
    \begin{itemize}
        \item Model Hamiltonian
        \sequ{
        H = -\sum_{\anglelr{ij}}J_{\alpha_{ij}}\dimer{i}{j} \quad ;\quad\wordfig{1.2}{figures/direction-def.pdf}
        }
        \item Majorana fermion (parton) representation: $b^x$, $b^y$, $b^z$, $c$ $\Rightarrow$  $\sigma^\mu = ib^\mu c$
        \item Parton Hamiltonian: define $ib^{\alpha_{ij}}_ib^{\alpha_{ij}}_j = \hat{u}_{ij}$
        \sequ{
        H \to \sum_{\anglelr{ij}}iJ_{\alpha_{ij}}\hat{u}_{ij}c_ic_j
        }
        \item Gutzwiller projection
        {\footnotesize
        \sequ{
        \mathcal{P}_G=\prod_{i}\lr{\frac{\mathds{1}+D_i}{2}} \quad,\quad D_i =  b^x_ib^y_ib^z_ic_i \Leftarrow -i\sigma^x_i\sigma^y_i\sigma^z_i
        }
        }
            \vspace{-2em}
        \begin{figure}[H]
            \centering
            \includegraphics[width=0.3\linewidth]{figures/gutzwiller_illu.pdf}
        \end{figure}

        \item Separability of the entanglement entropy \scite{https://doi.org/10.1103/PhysRevLett.105.080501}{(Yao and Qi, PRL\textbf{105} 080501)}
        \[
        \boxed{S_A = S_A^F + S_A^G}
        \]
    \end{itemize}
\end{frame}

\begin{frame}{Kitaev Model: Gauge Theory}
In the Majorana fermion representation, the model is equivalent to a system of free Majorana fermions (matter field) coupled to a static $\mathbb{Z}_2$ gauge field
\begin{itemize}

    \item Define the \textbf{gauge field} $\hat{u}_{ij} = e^{i \alert{\boldsymbol{A_{ij}}}}$, then the elementary \textbf{$\mathbb{Z}_2$ gauge flux} reads 
    \sequ{
    e^{i\Phi} = \exp\lr{i\oint_{\de p} A} = \prod_{\anglelr{ij}\in\de p}\hat{u}_{ij} = \prod_{\anglelr{ij}\in\de p}\dimer{i}{j} =: W_p
    }
    
    \item The \textbf{electric field} in the gauge field that flips the gauge field is $\boxed{b^{\alpha_{ij}}_i \to e^{i\pi \valert{\boldsymbol{E_{ij}}}}}$ because
    $b^{\alpha_{ij}}_i\hat{u}_{ij}b^{\alpha_{ij}}_i = -\hat{u}_{ij} \Longrightarrow [\alert{\boldsymbol{A_{ij}}}, \valert{\boldsymbol{E_{i'j'}}}] = i\delta_{ii'}\delta_{jj'}$

    \item \textbf{Gutzwiller projection} as a \textbf{Gauss law} constraint: $D_i = b_i^xb_i^yb_i^zc_i = 1$. The ground state is
    \[
        \prod_{i}\lr{\frac{\mathds{1}+D_i}{2}}\ket{u}\ket{M} \quad;\quad \text{Fix the gauge: $\bra{u}\hat{u}_{ij}\ket{u}=+1\forall\anglelr{ij}$}
    \]

    %%%%%%%%%%%%%%%% problems %%%%%%%%%%%%%%%%

    % \item The Gauss law is a physical constraint and cannot be broken.

\end{itemize}

\end{frame}

\begin{frame}{The 1st Layer Organization}
    \begin{itemize}
        \item<1-> The density matrix of a spin-1/2 system can be expanded in terms of Pauli-group elements
        \[
        \rho \propto \sum_{\mathsf{S}\in \mathcal{P}_N} \anglelr{\mathsf{S}} \mathsf{S}
        \]

        \item<1-> The string operators $\Sigma_\mathfrak{D}$ in the spin-$1/2$ Kitaev model are Pauli strings up to a sign.
        \[
        \rho 
        \propto 
        \sum_{\mathfrak{D}} \mathfrak{C}_\mathfrak{D} \Sigma_{\mathfrak{D}}
        \]
        
        \item<2-> Strings belonging to the same class can be grouped.
        \[

            \mathfrak{C}_{\mathfrak{D}_1} \Sigma_{\mathfrak{D}_1} + \mathfrak{C}_{\mathfrak{D}_2} \Sigma_{\mathfrak{D}_2}

        \[

    \end{itemize}
\end{frame}
